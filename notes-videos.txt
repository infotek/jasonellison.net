# notes on creating low rez videos with an nvidia

cd ./jasonellison.net/html/video

# 1. MP4 NVENC (640x360 @ 24fps)
ffmpeg -hwaccel cuda -i Birdiefish_run_with_me_in_on_jib_sheets.mp4 \
  -vf "scale=-2:360,fps=24" \
  -c:v h264_nvenc -cq 32 -preset p6 \
  -c:a aac -b:a 96k \
  -movflags +faststart \
  Birdiefish_retro.mp4

# 2. WebM VP9 (640x360 @ 24fps)
ffmpeg -i Birdiefish_run_with_me_in_on_jib_sheets.mp4 \
  -vf "scale=-2:360,fps=24" \
  -c:v libvpx-vp9 -crf 40 -b:v 0 \
  -row-mt 1 -threads 16 \
  -c:a libopus -b:a 96k \
  Birdiefish_retro.webm

# 3. Make a poster for pre-load

ffmpeg -ss 00:00:03 -i video/Birdiefish_run_with_me_in_on_jib_sheets.mp4 \
  -vframes 1 -q:v 2 \
  video/poster.jpg
